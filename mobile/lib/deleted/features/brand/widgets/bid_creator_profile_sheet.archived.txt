import 'package:flutter/material.dart';
import 'package:go_router/go_router.dart';

import '../../../core/router/route_names.dart';
import '../../../core/theme/app_colors.dart';
import '../../../core/theme/app_shadows.dart';
import '../../../core/theme/app_spacing.dart';
import '../../../core/theme/app_typography.dart';
import '../../../providers/brand_provider.dart';
import '../../../widgets/buttons/bc_primary_button.dart';
import '../../../widgets/cards/bc_card.dart';

void openBidCreatorProfileSheet(
  BuildContext context, {
  required CampaignBid bid,
  String? displayName,
  String? avatarUrl,
}) {
  showModalBottomSheet<void>(
    context: context,
    isScrollControlled: true,
    backgroundColor: Colors.transparent,
    builder: (_) => _BidCreatorProfileSheet(
      bid: bid,
      displayName: displayName,
      avatarUrl: avatarUrl,
    ),
  );
}

class _BidCreatorProfileSheet extends StatelessWidget {
  const _BidCreatorProfileSheet({
    required this.bid,
    this.displayName,
    this.avatarUrl,
  });

  final CampaignBid bid;
  final String? displayName;
  final String? avatarUrl;

  @override
  Widget build(BuildContext context) {
    final resolvedAvatarUrl = (avatarUrl ?? '').trim().isNotEmpty
        ? avatarUrl!.trim()
        : (bid.creatorAvatarUrl ?? '').trim();
    final fullName = (displayName ?? '').trim().isNotEmpty
        ? displayName!.trim()
        : bid.creatorFullName;

    return DraggableScrollableSheet(
      initialChildSize: 0.7,
      maxChildSize: 1,
      minChildSize: 0.55,
      expand: false,
      builder: (context, controller) {
        return Container(
          decoration: const BoxDecoration(
            color: AppColors.surface,
            borderRadius: BorderRadius.vertical(top: Radius.circular(20)),
          ),
          child: ListView(
            controller: controller,
            padding: const EdgeInsets.only(bottom: AppSpacing.huge),
            children: [
              Container(
                width: double.infinity,
                padding: const EdgeInsets.fromLTRB(
                  AppSpacing.screenPadding,
                  AppSpacing.lg,
                  AppSpacing.screenPadding,
                  AppSpacing.lg,
                ),
                decoration: BoxDecoration(
                  color: const Color(0xFF0D0D0D),
                  borderRadius: const BorderRadius.vertical(
                    top: Radius.circular(20),
                  ),
                ),
                child: Column(
                  children: [
                    Container(
                      width: 72,
                      height: 72,
                      decoration: BoxDecoration(
                        shape: BoxShape.circle,
                        border: Border.all(
                          color: AppColors.neon,
                          width: 3,
                        ),
                      ),
                      child: CircleAvatar(
                        radius: 34,
                        backgroundColor: AppColors.surface,
                        backgroundImage: resolvedAvatarUrl.isNotEmpty
                            ? NetworkImage(resolvedAvatarUrl)
                            : null,
                        child: resolvedAvatarUrl.isNotEmpty
                            ? null
                            : Text(
                                bid.creatorInitials,
                                style: AppTypography.displaySM,
                              ),
                      ),
                    ),
                    const SizedBox(height: AppSpacing.sm),
                    Text(
                      fullName.trim().isNotEmpty
                          ? fullName
                          : bid.creatorFirstName,
                      style: AppTypography.labelLG
                          .copyWith(color: AppColors.white),
                    ),
                    const SizedBox(height: AppSpacing.xs),
                    Text(
                      bid.creatorHandle.trim().isNotEmpty
                          ? '@${bid.creatorHandle}'
                          : 'Creator',
                      style: AppTypography.bodySM
                          .copyWith(color: AppColors.darkMuted),
                    ),
                  ],
                ),
              ),
              const SizedBox(height: AppSpacing.md),
              Padding(
                padding: const EdgeInsets.symmetric(
                  horizontal: AppSpacing.screenPadding,
                ),
                child: BCCard(
                  shadow: AppShadows.neon,
                  child: Row(
                    children: [
                      Expanded(
                          child: _miniStat('Followers',
                              _compactFollowers(bid.followersCount))),
                      const SizedBox(width: AppSpacing.sm),
                      Expanded(
                          child: _miniStat('Tier', bid.tier.toUpperCase())),
                    ],
                  ),
                ),
              ),
              const SizedBox(height: AppSpacing.md),
              Padding(
                padding: const EdgeInsets.symmetric(
                  horizontal: AppSpacing.screenPadding,
                ),
                child: BCCard(
                  shadow: AppShadows.pink,
                  child: Column(
                    crossAxisAlignment: CrossAxisAlignment.start,
                    children: [
                      const Text('About', style: AppTypography.labelMD),
                      const SizedBox(height: AppSpacing.sm),
                      Text(
                        bid.niches.isNotEmpty
                            ? bid.niches.join(', ')
                            : 'No additional profile details available yet.',
                        style: AppTypography.bodySM,
                      ),
                    ],
                  ),
                ),
              ),
              const SizedBox(height: AppSpacing.lg),
              Padding(
                padding: const EdgeInsets.symmetric(
                  horizontal: AppSpacing.screenPadding,
                ),
                child: BCPrimaryButton.dark(
                  label: 'View Full Profile ->',
                  onPressed: () {
                    Navigator.of(context).pop();
                    context.push(
                      RouteNames.partnerProfile,
                      extra: {
                        'id': bid.creatorId,
                        'first_name': bid.creatorFirstName,
                        'full_name': fullName.trim().isNotEmpty
                            ? fullName
                            : bid.creatorFullName,
                        'name': fullName.trim().isNotEmpty
                            ? fullName
                            : bid.creatorFirstName,
                        'avatar_url': resolvedAvatarUrl,
                        'avatarUrl': resolvedAvatarUrl,
                        'role': 'creator',
                        'viewerRole': 'brand',
                        'tier': bid.tier,
                        'bio': bid.niches.join(', '),
                        'followers': bid.followersCount,
                        'readOnly': true,
                      },
                    );
                  },
                ),
              ),
            ],
          ),
        );
      },
    );
  }

  Widget _miniStat(String label, String value) {
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Text(label,
            style: AppTypography.bodyXS.copyWith(color: AppColors.muted)),
        const SizedBox(height: AppSpacing.xs),
        Text(value, style: AppTypography.labelMD),
      ],
    );
  }

  String _compactFollowers(int followers) {
    if (followers >= 1000000) {
      final value = followers / 1000000;
      final str =
          value % 1 == 0 ? value.toStringAsFixed(0) : value.toStringAsFixed(1);
      return '${str}m';
    }
    if (followers >= 1000) {
      final value = followers / 1000;
      final str =
          value % 1 == 0 ? value.toStringAsFixed(0) : value.toStringAsFixed(1);
      return '${str}k';
    }
    return followers.toString();
  }
}
